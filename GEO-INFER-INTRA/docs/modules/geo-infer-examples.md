# GEO-INFER-EXAMPLES: Cross-Module Examples

> **Purpose**: Cross-module integration demonstrations and tutorials
>

## Overview
Note: All examples are runnable; see `GEO-INFER-EXAMPLES/examples` for scripts.

```python # Tutorial 1: Basic Spatial Analysis from geo_infer_space import SpatialAnalyzer from geo_infer_data import DataLoader # Load sample data loader = DataLoader() points = loader.load_sample('cities_usa') polygons = loader.load_sample('states_usa') # Initialize spatial analyzer analyzer = SpatialAnalyzer( coordinate_system='EPSG:4326', h3_resolution=7 ) # Perform spatial operations # Point-in-polygon cities_by_state = analyzer.spatial_join( points=points, polygons=polygons, how='inner' ) # Calculate distances distances = analyzer.calculate_distances( from_points=points, to_points=points, method='haversine' ) # Create H3 hexagons hexagons = analyzer.create_h3_grid( boundary=polygons.unary_union, resolution=5 ) # Aggregate to hexagons population_hex = analyzer.aggregate_to_h3( data=points, values='population', aggregation='sum' ) ```
 ```python # Tutorial 2: Time Series Analysis from geo_infer_time import TemporalAnalyzer from geo_infer_data import DataLoader # Load time series data loader = DataLoader() timeseries = loader.load_sample('temperature_daily') # Initialize temporal analyzer temporal = TemporalAnalyzer( frequency='daily', timezone='UTC' ) # Decompose time series decomposition = temporal.decompose( data=timeseries, method='stl', seasonal_period=365 ) # Detect trends trends = temporal.detect_trends( data=timeseries, method='mann_kendall', confidence=0.95 ) # Forecast future values forecast = temporal.forecast( data=timeseries, horizon=30, method='prophet', include_uncertainty=True ) ``` ```python # Tutorial 3: Active Inference Basics from geo_infer_act import ActiveInferenceModel from geo_infer_data import DataLoader # Load sample spatial data loader = DataLoader() observations = loader.load_sample('sensor_readings') # Initialize Active Inference model model = ActiveInferenceModel( state_space=['location', 'activity'], observation_space=['gps', 'accelerometer'], action_space=['stay', 'move'] ) # Set prior beliefs model.set_priors( beliefs={'location': 'home', 'activity': 'resting'}, precision=1.0 ) # Update with observations for obs in observations: model.update_beliefs(obs) action = model.select_action() model.execute_action(action) # Get posterior beliefs posterior = model.get_posterior() ``` ### 2. Domain-Specific Use Cases **Purpose**: solutions for specific application domains. ```python # Use Case: Environmental Monitoring System from geo_infer_space import SpatialAnalyzer from geo_infer_climate import ClimateAnalyzer from geo_infer_water import WaterQualityAnalyzer from geo_infer_bio import BiodiversityAssessor from geo_infer_iot import SensorNetwork from geo_infer_act import ActiveInferenceModel class EnvironmentalMonitoringSystem: """environmental monitoring system using GEO-INFER modules.""" def __init__(self, study_area): self.study_area = study_area self.spatial = SpatialAnalyzer() self.climate = ClimateAnalyzer() self.water = WaterQualityAnalyzer() self.bio = BiodiversityAssessor() self.sensors = SensorNetwork() self.ai = ActiveInferenceModel() def setup_monitoring_network(self, sensor_locations): """Deploy monitoring sensor network.""" self.sensors.configure( locations=sensor_locations, parameters=['temperature', 'humidity', 'water_quality', 'air_quality'], sampling_interval='15min' ) def analyze_current_conditions(self): """Analyze current environmental conditions.""" # Get sensor readings readings = self.sensors.get_current_readings() # Spatial interpolation conditions_map = self.spatial.interpolate( points=readings, method='kriging', resolution=100 ) # Identify anomalies anomalies = self.ai.detect_anomalies( observations=readings, expected_model=self.ai.generative_model ) return conditions_map, anomalies def assess_ecosystem_health(self): """Assess ecosystem health from monitoring data.""" # Water quality assessment water_quality = self.water.assess( sensors=self.sensors.get_water_sensors(), standards='ecosystem_health' ) # Biodiversity indicators biodiversity = self.bio.calculate_indicators( observation_data=field_observations, spatial_resolution=self.study_area ) # Climate stress assessment climate_stress = self.climate.assess_stress( current_conditions=self.analyze_current_conditions()[0], baseline=historical_baseline ) # Integrated health index health_index = self.bio.calculate_health_index( components={ 'water_quality': water_quality, 'biodiversity': biodiversity, 'climate_stress': climate_stress } ) return health_index def predict_future_conditions(self, horizon_days): """Predict future environmental conditions.""" # Climate projections climate_forecast = self.climate.forecast( location=self.study_area, horizon_days=horizon_days ) # Active Inference predictions self.ai.update_beliefs(self.sensors.get_current_readings()) predictions = self.ai.predict_future_states( horizon_steps=horizon_days * 4 # 6-hour steps ) return predictions def generate_alerts(self): """Generate alerts for concerning conditions.""" conditions, anomalies = self.analyze_current_conditions() alerts = [] for anomaly in anomalies: if anomaly['severity'] >= 0.7: alerts.append({ 'type': anomaly['type'], 'location': anomaly['location'], 'severity': anomaly['severity'], 'recommended_action': self.ai.recommend_action(anomaly) }) return alerts ``` ```python # Best Practices **Purpose**: Document recommended patterns and anti-patterns. ```python # Best Practice 1: Efficient Data Loading from geo_infer_data import DataLoader # GOOD: Use streaming for large datasets loader = DataLoader() for chunk in loader.stream_large_file( path='large_dataset.parquet', chunk_size=10000 ): process_chunk(chunk) # BAD: Loading entire large file into memory # entire_file = loader.load('large_dataset.parquet') # May cause MemoryError # Best Practice 2: Coordinate System Management from geo_infer_space import SpatialAnalyzer # GOOD: Consistent CRS throughout pipeline analyzer = SpatialAnalyzer(coordinate_system='EPSG:4326') data = analyzer.ensure_crs(input_data, target_crs='EPSG:4326') result = analyzer.analyze(data) # BAD: Mixing coordinate systems without transformation # result = analyzer.analyze(data_in_epsg_3857) # CRS mismatch errors # Best Practice 3: Error Handling from geo_infer_space import SpatialAnalyzer # GOOD: Proper error handling with context try: result = analyzer.spatial_join(points, polygons) except GeometryError as e: logger.error(f"Geometry error in spatial join: {e}") result = analyzer.repair_geometries(points, polygons) result = analyzer.spatial_join(result['points'], result['polygons']) # Best Practice 4: Resource Management from contextlib import contextmanager # GOOD: Use context managers for resources @contextmanager def spatial_session(): analyzer = SpatialAnalyzer() analyzer.initialize_cache() try: yield analyzer finally: analyzer.clear_cache() with spatial_session() as analyzer: result = analyzer.analyze(data) # Best Practice 5: Parallel Processing from geo_infer_space import SpatialAnalyzer # GOOD: Use built-in parallel processing analyzer = SpatialAnalyzer() analyzer.enable_parallel_processing(n_workers=8) results = analyzer.batch_analyze(datasets) # BAD: Manual multiprocessing that bypasses geo-infer optimizations # with multiprocessing.Pool() as pool: # results = pool.map(analyze_one, datasets) ``` ### 5.
